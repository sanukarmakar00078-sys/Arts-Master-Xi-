<script>
// Welcome splash and tap-first jelly feedback: run the animation before activating controls.
(function(){
  const splash = document.getElementById('artsWelcomeSplash');
  window.setTimeout(() => { if (splash) splash.remove(); }, 2750);
  let replaying = false;
  // Bottom iOS dock tabs keep the original quick Jelly feedback; do not delay their navigation.
  const selector = 'button:not(.dock-btn), a, [role="button"]:not(.dock-btn), .card, .topic, .option, .note-pill, .history-item, .setting-row, .share-icon-btn, .admin-tab';
  document.addEventListener('click', function(event){
    if (replaying) return;
    const target = event.target && event.target.closest ? event.target.closest(selector) : null;
    if (!target || target.disabled || target.closest('#artsWelcomeSplash')) return;
    // Home / Notes / Profile dock buttons must respond immediately with their original short dock animation.
    if (target.classList.contains('dock-btn') || target.closest('.bottom-dock-wrap')) return;
    // Let text fields and form controls retain their normal behavior.
    if (target.matches('input, select, textarea, label')) return;
    event.preventDefault(); event.stopPropagation(); event.stopImmediatePropagation();
    target.classList.remove('tap-jelly-playing');
    void target.offsetWidth;
    target.classList.add('tap-jelly-playing');
    window.setTimeout(() => {
      target.classList.remove('tap-jelly-playing');
      replaying = true;
      target.dispatchEvent(new MouseEvent('click', {bubbles:true, cancelable:true, view:window}));
      replaying = false;
    }, 500);
  }, true);
})();
</script>

<script>
(function(){
  const classScreen = document.getElementById('classSelectScreen');
  const semesterScreen = document.getElementById('classNextScreen');
  const comingScreen = document.getElementById('comingSoonScreen');
  const existingApp = document.getElementById('existingApp');

  function show(screen){
    document.body.classList.toggle('class-screen-active', screen !== 'app');
    classScreen.style.display = screen === 'classes' ? 'flex' : 'none';
    semesterScreen.style.display = screen === 'semester' ? 'flex' : 'none';
    comingScreen.style.display = screen === 'soon' ? 'flex' : 'none';
    existingApp.style.display = screen === 'app' ? 'block' : 'none';
    window.scrollTo(0,0);

  document.querySelectorAll('.class-choice').forEach(button => {
    button.addEventListener('click', () => {
      const selectedClass = button.dataset.class;

  async function publishSharedUpdate(alertMessage = "") {
    if (alertMessage) {
      appData.latestAlert = {
        id: "alert_" + Date.now() + "_" + Math.random().toString(36).slice(2, 6),
        message: alertMessage,
        time: new Date().toLocaleTimeString()
      };
      lastSeenAlertId = appData.latestAlert.id;
      try { localStorage.setItem("arts_master_xi_seen_alert", lastSeenAlertId); } catch (_) {}
    }

    if (bcast) {
      try { bcast.postMessage({ type: "STATE_UPDATE", payload: appData }); } catch (_) {}
    }

    if (sharedDocRef) {
      try {
        await setDoc(sharedDocRef, appData, { merge: false });
        setSyncPill("🟢 Firebase Synced");
      } catch (error) {
        console.error("Firestore save failed:", error);
        setSyncPill("🔒 Firestore permission needed");
        toast("Cloud save হয়নি। Firebase Authentication ও Firestore Rules সেট করতে হবে।");
      }
    } else {
      setSyncPill("⚠️ Firebase unavailable");
    }
  }

  if (sharedDocRef) {
    onSnapshot(sharedDocRef, snap => {
      if (snap.exists()) {
        applySharedPayload(snap.data(), true);
        setSyncPill("🟢 Firebase Synced");
      } else {
        setSyncPill("☁️ Firebase ready — no cloud data yet");
      }
    }, error => {

  function playAppSettingTone() {
    if (!appSettings.sound || Number(appSettings.volume) <= 0) return;
    try { const Ctx = window.AudioContext || window.webkitAudioContext; if (!Ctx) return;
      const ctx = new Ctx(), osc = ctx.createOscillator(), gain = ctx.createGain();
      osc.type = "sine"; osc.frequency.value = 660;
      gain.gain.setValueAtTime((Number(appSettings.volume) / 100) * 0.045, ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + 0.09);
      osc.connect(gain); gain.connect(ctx.destination); osc.start(); osc.stop(ctx.currentTime + 0.1);
      osc.onended = () => ctx.close();
    } catch (_) {}
